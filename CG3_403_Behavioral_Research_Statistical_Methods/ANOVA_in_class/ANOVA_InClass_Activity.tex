\section{Overview of Steps (from
slide)}\label{overview-of-steps-from-slide}

\begin{enumerate}
\def\labelenumi{\arabic{enumi}.}
\tightlist
\item
  Check for normality\\
\item
  Check homogeneity of variances → \textbf{Factorial / Independent
  ANOVA}\\
  Check sphericity of variances → \textbf{Repeated Measures ANOVA}\\
\item
  Choose the appropriate test\\
\item
  Only if F is significant → run post-hoc test\\
\item
  Report effect size\\
\item
  Plot the data
\end{enumerate}
